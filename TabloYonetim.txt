CREATE TABLE Roller (
    RolId INT IDENTITY(1,1) NOT NULL,
    RolAdi VARCHAR(50) NOT NULL,
    AktifMi BIT NOT NULL CONSTRAINT DF_Roller_AktifMi DEFAULT 1,

    CONSTRAINT PK_Roller PRIMARY KEY (RolId),
    CONSTRAINT UQ_Roller_RolAdi UNIQUE (RolAdi)
);
CREATE TABLE Yetkiler (
    YetkiId INT IDENTITY(1,1) NOT NULL,
    YetkiKodu VARCHAR(50) NOT NULL,
    YetkiAdi VARCHAR(100) NOT NULL,

    CONSTRAINT PK_Yetkiler PRIMARY KEY (YetkiId),
    CONSTRAINT UQ_Yetkiler_YetkiKodu UNIQUE (YetkiKodu)
);
CREATE TABLE PersonelRolleri (
    PersonelId INT NOT NULL,
    RolId INT NOT NULL,

    CONSTRAINT PK_PersonelRolleri PRIMARY KEY (PersonelId, RolId),

    CONSTRAINT FK_PersonelRolleri_Personel
        FOREIGN KEY (PersonelId) REFERENCES Personeller(PersonelId),

    CONSTRAINT FK_PersonelRolleri_Rol
        FOREIGN KEY (RolId) REFERENCES Roller(RolId)
);
CREATE TABLE RolYetkileri (
    RolYetkiId INT IDENTITY(1,1) NOT NULL,
    RolId INT NOT NULL,
    YetkiId INT NOT NULL,

    CONSTRAINT PK_RolYetkileri PRIMARY KEY (RolYetkiId),

    CONSTRAINT UQ_RolYetkileri UNIQUE (RolId, YetkiId),

    CONSTRAINT FK_RolYetkileri_Rol
        FOREIGN KEY (RolId) REFERENCES Roller(RolId),

    CONSTRAINT FK_RolYetkileri_Yetki
        FOREIGN KEY (YetkiId) REFERENCES Yetkiler(YetkiId)
);
